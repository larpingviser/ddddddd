-- EchoHttpGet: executor-agnostic HTTP GET used by Server Hop.
local function EchoHttpGet(url)
    local req = (syn and syn.request)
             or (http and http.request)
             or (fluxus and fluxus.request)
             or http_request
             or request
    if req then
        local ok, res = pcall(req, {Url = url, Method = "GET"})
        if ok and type(res) == "table" then
            if res.Body and res.Body ~= "" then return res.Body end
            if res.body and res.body ~= "" then return res.body end
        end
    end
    if game.HttpGetAsync then
        local ok, body = pcall(function() return game:HttpGetAsync(url) end)
        if ok and type(body) == "string" and body ~= "" then return body end
    end
    if game.HttpGet then
        local ok, body = pcall(function() return game:HttpGet(url) end)
        if ok and type(body) == "string" and body ~= "" then return body end
    end
    return nil
end

local function shuffledCopy(list)

    local copy = {}

    for i, v in ipairs(list) do copy[i] = v end

    for i = #copy, 2, -1 do

        local j = math.random(1, i)

        copy[i], copy[j] = copy[j], copy[i]

    end

    return copy

end

local function isEchoVisible()
    if Library.Unloaded then return false end

    if Library.Open ~= nil then
        return Library.Open and true or false
    end

    local root = Window.Holder or Window.Root or Window.Main or Window.Frame
    if root and root:IsA("GuiObject") then
        if not root.Visible then return false end
        local parent = root.Parent
        while parent and parent:IsA("GuiObject") do
            if not parent.Visible then return false end
            parent = parent.Parent
        end
    end
    return true
end

local function findAllRoots()

    local roots = {}

    local ok, gethui = pcall(function() return gethui() end)

    if ok and gethui then table.insert(roots, gethui) end

    pcall(function() table.insert(roots, game:GetService("CoreGui")) end)

    if Window.Holder then table.insert(roots, Window.Holder) end

    if Window.Root then table.insert(roots, Window.Root) end

    if Window.Main then table.insert(roots, Window.Main) end

    if Window.Frame then table.insert(roots, Window.Frame) end

    return roots

end

        end

            break

        end

        task.wait(0.05)

    end

    if not footerLabel then

        warn("[Z] Footer label not found")

        return

    end

    local CHAR_DELAY = 0.05

    local CYCLE_TIME = 25

    while footerLabel.Parent and not Library.Unloaded do

        -- Pause typing entirely while the menu is hidden
        if not isEchoVisible() then
            task.wait(0.25)
            continue
        end

        local queue = shuffledCopy(footerMessages)

        for _, msg in ipairs(queue) do

            if not footerLabel.Parent then break end

            -- Bail out mid-typing if the menu gets closed
            if not isEchoVisible() then break end

            footerLabel.Text = ""

            for i = 1, #msg do

                if not isEchoVisible() then break end

                footerLabel.Text = msg:sub(1, i)

                playTypeTick()

                task.wait(CHAR_DELAY)

            end

            local forwardTime = #msg * CHAR_DELAY

            local eraseTime = #msg * CHAR_DELAY

            local holdTime = CYCLE_TIME - forwardTime - eraseTime

            if holdTime > 0 then

                task.wait(holdTime)

            end

            for i = #msg - 1, 0, -1 do

                if not isEchoVisible() then break end

                footerLabel.Text = msg:sub(1, i)

                playTypeTick()

                task.wait(CHAR_DELAY)

            end

        end

    end

end)

local function GetEchoGreeting()

    local Hour = tonumber(os.date("%H")) or 12

    local Period

    if Hour >= 5 and Hour < 12 then

        Period = "Morning"

    elseif Hour >= 12 and Hour < 17 then

        Period = "Noon"

    elseif Hour >= 17 and Hour < 21 then

        Period = "Evening"

    else

        Period = "Night"

    local DisplayName = LocalPlayer.DisplayName or LocalPlayer.Name

    return Period .. ", " .. DisplayName .. ".\n\nWelcome to Echo Hub, The Owner and Developer of this hub is " .. DisplayName .. ". Hope you enjoy using the Script :)"

end

ProfileGroup:AddButton({

    Text = "Server Hop",

    Func = function()

        local TeleportService = game:GetService("TeleportService")
        local HttpService = game:GetService("HttpService")
        local Stats = game:GetService("Stats")

        Library:Notify({
            Title = "Echo",
            Description = "Finding the best server (low ping, high players)...",
            Icon = "bell",
            Time = 3,
        })

        task.spawn(function()
            -- ── 1. Best-effort current region ping ──────────────────
            local currentPing
            pcall(function()
                local item = Stats.Network.ServerStatsItem["Data Ping"]
                local value = item and item:GetValue()
                if type(value) == "number" and value > 0 then
                    currentPing = value
                end
            end)

            -- ── 2. Page through public servers (up to ~500) ─────────
            local servers = {}
            local seen = {}
            local cursor = ""

            for _ = 1, 5 do
                local url = "https://games.roblox.com/v1/games/"
                    .. tostring(game.PlaceId)
                    .. "/servers/Public?sortOrder=Asc&limit=100"
                if cursor ~= "" then
                    url = url .. "&cursor=" .. HttpService:UrlEncode(cursor)
                end

                local body
                pcall(function() body = EchoHttpGet(url) end)
                if type(body) ~= "string" or body == "" then break end

                local decodeOk, data = pcall(function()
                    return HttpService:JSONDecode(body)
                end)
                if not decodeOk or type(data) ~= "table" then break end

                for _, server in ipairs(data.data or {}) do
                    local id = server.id
                    local playing = tonumber(server.playing) or 0
                    local maxPlayers = tonumber(server.maxPlayers) or 0
                    if id and id ~= game.JobId
                        and playing < maxPlayers          -- must have space
                        and playing >= 2                  -- skip near-empty
                        and not seen[id] then
                        seen[id] = true
                        table.insert(servers, server)
                    end
                end

                cursor = data.nextPageCursor or ""
                if cursor == "" then break end
            end

            if #servers == 0 then
                Library:Notify({
                    Title = "Echo",
                    Description = "Server hop failed: no available server found.",
                    Icon = "bell",
                    Time = 3,
                })
                return
            end

            -- ── 3. Score each server ────────────────────────────────
            -- Priority (lower score = better):
            --   • Low average ping  (server.ping when provided)
            --   • More players (popular servers = better FPS/health)
            --   • Closer to our current ping (region proximity)
            local MAX_PLAYERS = Players.MaxPlayers
            local ranked = {}
            for _, server in ipairs(servers) do
                local serverPing  = tonumber(server.ping)
                local playing     = tonumber(server.playing) or 0
                local maxPlayers  = tonumber(server.maxPlayers) or MAX_PLAYERS
                local fillRatio   = maxPlayers > 0 and (playing / maxPlayers) or 0

                -- Ping term: absolute ping if known, else fallback to diff
                local pingTerm
                if serverPing then
                    pingTerm = serverPing
                    if currentPing then
                        -- 50% weight on absolute ping, 50% on delta from us
                        pingTerm = (serverPing * 0.5) + (math.abs(serverPing - currentPing) * 0.5)
                    end
                else
                    -- No ping data: neutral value so player-count decides
                    pingTerm = 120
                end

                -- Player term: reward fuller servers, slight reward for very full
                -- (FPS tends to be similar across lobbies in Roblox; player
                --  count is the closest proxy we can get to "active" servers).
                -- Normalize to 0..100.
                local playerTerm = (1 - fillRatio) * 100

                -- Final weighted score
                local score = (pingTerm * 0.55) + (playerTerm * 0.45)

                -- Small random jitter so repeated hops don't always pick the same
                score = score + math.random() * 5

                table.insert(ranked, {
                    Id = server.id,
                    Score = score,
                    Ping = serverPing,
                    Playing = playing,
                    Max = maxPlayers,
                })
            end

            table.sort(ranked, function(a, b)
                return a.Score < b.Score
            end)

            -- Pick randomly from the top 5 to add variety while still
            -- guaranteeing the choice is among the best.
            local poolSize = math.min(5, #ranked)
            local selected = ranked[math.random(1, poolSize)]
            if not selected or not selected.Id then return end

            local pingText = selected.Ping and (math.floor(selected.Ping) .. "ms") or "unknown ping"
            Library:Notify({
                Title = "Echo",
                Description = "Joining server — " .. selected.Playing .. "/" .. selected.Max
                    .. " players, " .. pingText,
                Icon = "bell",
                Time = 3,
            })

            task.wait(0.2)

            -- ── 4. Teleport (with retry + queue fallback) ───────────
            local teleported = false
            local ok = pcall(function()
                TeleportService:TeleportToPlaceInstance(game.PlaceId, selected.Id, LocalPlayer)
                teleported = true
            end)

            if not ok or not teleported then
                Library:Notify({
                    Title = "Echo",
                    Description = "Server hop teleport failed.",
                    Icon = "bell",
                    Time = 3,
                })
            end
        end)

    end,

})

        Frames += 1

        local Elapsed = Now - LastFPSUpdate

            local CurrentFPS = math.floor(

        local Seconds = math.floor(

        local Minutes = math.floor((Seconds % 3600) / 60)

        local Remaining = Seconds % 60

        SessionLabel:SetText(string.format(

            "Session time: %02d:%02d:%02d",

            Hours,

            Minutes,

            Remaining

        ))

        local Ping

        pcall(function()

            Ping = Stats

                .Network

                .ServerStatsItem["Data Ping"]

                :GetValue()

        end)

        if Ping and PingLabel and PingLabel.Parent then

            PingLabel:SetText(

                "Ping: " .. math.floor(Ping + 0.5) .. " ms"

            )

        end

        task.wait(0.5)

    end

    pcall(function()

        FPSConnection:Disconnect()

    end)

        local Distance = (TargetPos - CurrentPos).Magnitude

            Root.CFrame = Target
            Root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            Root.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
            return
        end
        local Alpha = math.clamp(LoopTPSmoothSpeed * dt * 60, 0, 1)
        local NewCFrame = Root.CFrame:Lerp(Target, Alpha)
        Root.CFrame = NewCFrame
        Root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        Root.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
    end)

local function TBFindTarget()
    local c = LocalChar()
    if not c or not c:FindFirstChild('HumanoidRootPart') then return end
    if SvcWS:FindFirstChild('GrabParts') then return end
    local origin = Cam.CFrame.Position
    local dir = Cam.CFrame.LookVector
    TBRayPrms.FilterDescendantsInstances = { c, SvcWS.Terrain }
    local res
    for _, d in ipairs({ dir, (dir + Vector3.new(0, 0.075, 0)).Unit, (dir + Vector3.new(0, -0.075, 0)).Unit }) do
        res = SvcWS:Raycast(origin, d * 1000, TBRayPrms)
        if res then break end
    end
    if not res then return end
    local m = res.Instance:FindFirstAncestorOfClass('Model')
    if not m or m == c then return end
    local hum = m:FindFirstChildOfClass('Humanoid')
    local root = m:FindFirstChild('HumanoidRootPart')
    if not hum or hum.Health <= 0 or not root then return end
    if (c.HumanoidRootPart.Position - root.Position).Magnitude > TBDist then return end
    return m
end

local function ApplyTriggerBot(v)
    if TBConn then TBConn:Disconnect() TBConn = nil end
    if not v then return; end
    TBConn = SvcRun.Heartbeat:Connect(function()
        if not TBReady then return end
        if SvcInput:GetFocusedTextBox() then return end
        if tick() - TBLastCheck < TBThrottle then return end
        TBLastCheck = tick()
        local t = TBFindTarget()
        if t then
            TBLastTarget = t
            TBLastHit = tick()
        elseif TBLastTarget and tick() - TBLastHit > TBMemory then
            TBLastTarget = nil
        end
        local c = LocalChar()
        local root = TBLastTarget and TBLastTarget:FindFirstChild('HumanoidRootPart')
        if not (TBLastTarget and c and c:FindFirstChild('HumanoidRootPart') and root) then return end
        if (c.HumanoidRootPart.Position - root.Position).Magnitude > TBDist then
            TBLastTarget = nil
            return
        end
        TBReady = false
        task.spawn(function()
            task.wait(TBPreDelay)
            pcall(mouse1press)
            local t0 = tick()
            repeat task.wait(0.02) until not SvcWS:FindFirstChild('GrabParts') or tick() - t0 > 1.6
            task.wait(TBPostDelay)
            TBReady = true
            TBLastTarget = nil
        end)
    end)
end

local function ApplyExtendLine(v)
    infLineExtendT = v
    if extendInputConn then extendInputConn:Disconnect() extendInputConn = nil end
    if extendChildConn then extendChildConn:Disconnect() extendChildConn = nil end
    if not v then lineDistanceV = 11; return; end
    extendInputConn = SvcInput.InputChanged:Connect(function(input)
        if not infLineExtendT then return end
        if input.UserInputType == Enum.UserInputType.MouseWheel then
            if input.Position.Z > 0 then
                lineDistanceV = lineDistanceV + increaseLineExtendV
            elseif input.Position.Z < 0 then
                lineDistanceV = math.max(11, lineDistanceV - increaseLineExtendV);
            end
        end
    end)
    extendChildConn = SvcWS.ChildAdded:Connect(function(child)
        if not infLineExtendT then return end
        if child.Name ~= 'GrabParts' or not child:IsA('Model') then return end
        local dragPart = child:WaitForChild('DragPart')
        local cloned = dragPart:Clone()
        cloned.Name = 'DragPart1'
        local attach = cloned:FindFirstChild('DragAttach')
        if cloned:FindFirstChild('AlignPosition') and attach then
            cloned.AlignPosition.Attachment1 = attach
        end
        cloned.Parent = child
        if cloned:FindFirstChild('AlignOrientation') then
            cloned.AlignOrientation.Enabled = false
        end
        if dragPart:FindFirstChild('AlignPosition') then
            dragPart.AlignPosition.Enabled = false
        end
        task.spawn(function()
            while child.Parent and infLineExtendT do
                cloned.Position = Cam.CFrame.Position + Cam.CFrame.LookVector * lineDistanceV
                SvcRun.RenderStepped:Wait()
            end
            lineDistanceV = 11
        end)
    end)
end

local function ApplyKickGrab(v)
    if kg_mainThread then task.cancel(kg_mainThread) kg_mainThread = nil end
    if kg_inputConn then kg_inputConn:Disconnect() kg_inputConn = nil end
    if not v then kg_Active = false; return; end
    kg_Active = true
    kg_mainThread = task.spawn(function()
        while kg_Active do
            local grabParts = SvcWS:FindFirstChild('GrabParts')
            if not grabParts then task.wait(); continue; end
            local gp = grabParts:FindFirstChild('GrabPart')
            local weld = gp and gp:FindFirstChildOfClass('WeldConstraint')
            local part1 = weld and weld.Part1
            if part1 then
                local ownerPlayer = nil
                for _, pl in ipairs(SvcPlayers:GetPlayers()) do
                    if pl.Character and part1:IsDescendantOf(pl.Character) then
                        ownerPlayer = pl
                        break
                    end
                end
                if ownerPlayer then
                    while kg_Active and SvcWS:FindFirstChild('GrabParts') do
                        local tgtRoot = ownerPlayer.Character and ownerPlayer.Character:FindFirstChild('HumanoidRootPart')
                        local myRoot  = Me.Character and Me.Character:FindFirstChild('HumanoidRootPart')
                        local tgtHead = ownerPlayer.Character and ownerPlayer.Character:FindFirstChild('Head')
                        if tgtRoot and myRoot and tgtHead then
                            pcall(function()
                                EvSetOwner:FireServer(tgtRoot, CFrame.lookAt(myRoot.Position, tgtRoot.Position))
                            end)
                            task.wait()
                            pcall(function()
                                EvDestroyLine:FireServer(tgtHead)
                            end)
                        end
                        task.wait()
                    end
                end
            end
            task.wait()
        end
    end)
end

local function isToyPart(part)
    if not part or not part.Parent then return false end

    local ancestor = part.Parent

    while ancestor and ancestor ~= SvcWS do
        if ancestor:IsA("Folder") and ancestor.Name:match("SpawnedInToys$") then
            return true
        end
        if ancestor.Name == "PlotItems" then
            return true
        end
        ancestor = ancestor.Parent
    end

    return false
end

local function getAuraTargets(rootPos)
    local targets = {}
    local dist = Options.AuraDistance and Options.AuraDistance.Value or 20
    local sel  = AuraModes.Selection or "Players"

    if sel == "Players" or sel == "Both" then
        for _, player in pairs(SvcPlayers:GetPlayers()) do
            if player ~= Me and player.Character then
                local hrp = player.Character:FindFirstChild("HumanoidRootPart")
                if hrp and (hrp.Position - rootPos).Magnitude <= dist then
                    table.insert(targets, hrp)
                end
            end
        end
    end

    if sel == "Objects" or sel == "Both" then
        for _, obj in ipairs(SvcWS:GetPartBoundsInRadius(rootPos, dist)) do
            if obj:IsA("BasePart")
                and not obj.Anchored
                and not obj:IsDescendantOf(Me.Character)
                and isToyPart(obj)
            then
                table.insert(targets, obj)
            end
        end
    end

    return targets
end

local function runAuraLoop(key, perTarget)
    _G[key] = true
    if _G[key .. "_Thread"] then return end
    _G[key .. "_Thread"] = true
    task.spawn(function()
        while _G[key] do

            local root = char and char:FindFirstChild("HumanoidRootPart")
            if root then
                for _, target in ipairs(getAuraTargets(root.Position)) do
                    pcall(function()

                        perTarget(target)
                    end)

        end
        _G[key .. "_Thread"] = false
    end)
end

    -- Shared live anchor getter — used by Apply buttons AND the Loop runner.

    local function makeAnchorGetter(idx, total)
        return function()
            local hrp = TGetMyHRP()
            local basePos = hrp and hrp.Position or Vector3.new(0, 0, 0)
            local offsetAngle = (idx - 1) * (math.pi * 2 / math.max(total, 1))
            local offsetRadius = 6 + math.floor((idx - 1) / 8) * 4
            return basePos + Vector3.new(
                math.cos(offsetAngle) * offsetRadius + kickOffsetX,
                kickOffsetY + math.floor((idx - 1) / 4) * 5,
                math.sin(offsetAngle) * offsetRadius + kickOffsetZ
            )
        end
    end

    local function OK_NukeForeignWelds(char)
        if not char or not char.Parent then return end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        for _, d in ipairs(hrp:GetChildren()) do
            if d:IsA("Weld") or d:IsA("WeldConstraint") or d:IsA("Motor6D") then
                local p0, p1 = d.Part0, d.Part1
                if p0 and p1 and not (p0:IsDescendantOf(char) and p1:IsDescendantOf(char)) then
                    pcall(function() d:Destroy() end)
                end
            end
        end
        for _, d in ipairs(char:GetDescendants()) do
            if d:IsA("WeldConstraint") or d:IsA("Weld") then
                local p0, p1 = d.Part0, d.Part1
                if p0 and p1 and not (p0:IsDescendantOf(char) and p1:IsDescendantOf(char)) then
                    pcall(function() d:Destroy() end)
                end
            elseif d:IsA("BodyPosition") or d:IsA("BodyGyro")
                or d:IsA("BodyVelocity") or d:IsA("BodyAngularVelocity")
                or d:IsA("VectorForce") or d:IsA("LinearVelocity")
                or d:IsA("AngularVelocity") or d:IsA("AlignPosition")
                or d:IsA("AlignOrientation") then
                local nm = d.Name
                if not nm:find("EchoKick") then
                    local p = d.Parent
                    if p == hrp or (p and p:IsDescendantOf(char)) then
                        pcall(function() d:Destroy() end)
                    end
                end
            end
        end
    end

    local function OK_FindPCLD(target)
        if not target then return nil end

        for _, c in ipairs(char:GetChildren()) do
            if c.Name == "PlayerCharacterLocationDetector" and c:IsA("BasePart") then
                return c

        for _, c in ipairs(char:GetDescendants()) do
            if c.Name == "PlayerCharacterLocationDetector" and c:IsA("BasePart") then
                return c

        end
        local hrp = char:FindFirstChild("HumanoidRootPart")

    -- ═══════════════════════════════════════════════════════════════
    -- Global remote budget — prevents ping spike when locking many
    -- players at once. We cap total SetNetOwner fires per second across
    -- ALL targets, so latency stays low no matter how many locks are on.
    -- ═══════════════════════════════════════════════════════════════
    local RemoteBudget = {
        -- Max SetNetOwner fires across all targets per second.
        MaxPerSec = 220,
        -- Sliding 1-second window of recent fire counts.
        Window = {},
    }

    local function RemoteBudget_Take(n)
        n = n or 1
        local now = tick()
        -- Drop entries older than 1 second.
        for i = #RemoteBudget.Window, 1, -1 do
            if now - RemoteBudget.Window[i].t > 1.0 then
                table.remove(RemoteBudget.Window, i)
            else
                break
            end
        end
        -- Count what we've used in the last second.
        local used = 0
        for _, e in ipairs(RemoteBudget.Window) do used = used + e.n end
        local remaining = RemoteBudget.MaxPerSec - used
        if remaining <= 0 then return 0 end
        local allowed = math.min(n, remaining)
        table.insert(RemoteBudget.Window, {t = now, n = allowed})
        return allowed
    end

        if not myHRP then return nil end

        local spawned = nil
        local conn
        conn = inv.ChildAdded:Connect(function(child)
            if child.Name == "PalletLightBrown" then spawned = child end
        end)
        task.spawn(function()
            pcall(function()
                TSpawnToy:InvokeServer("PalletLightBrown", myHRP.CFrame * CFrame.new(0, 20, 20), Vector3.zero)
            end)
        end)
        local t0 = tick()

        if not sound then
            pcall(function() TDestroyToy:FireServer(spawned) end)

        end

        for _, v in ipairs(spawned:GetDescendants()) do
            if v:IsA("BasePart") then
                v.CanCollide   = false
                v.CanQuery     = false
                v.CanTouch     = false
                v.Transparency = 1
            end
        end

        local ref = { pallet = spawned, sound = sound }
        OwnershipKickPallet[name] = ref
        return ref
    end

--/(ownership core — Strong Ownership Kick, with live XYZ anchor offsets)
    local function RunOwnershipKick(targetName, keepGoing, anchorGetter)
        local myChar = TLP.Character
        local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
        if not myRoot then return end

        local savedPos      = myRoot.CFrame
        local savedAnchored = myRoot.Anchored
        -- Do NOT anchor ourselves; server rejects SetNetworkOwner when frozen.

        local mode = selectedOwnershipMode
        local cfg = cfgByMode[mode] or cfgByMode["Ownership kick (Strong)"]

        local grabStartTime  = 0
        local dragging       = false
        local checkStartTime = 0
        local ownedRoot      = nil

        while keepGoing() do
            myChar = TLP.Character
            myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
            if not myRoot then break end

            myRoot.AssemblyLinearVelocity  = Vector3.zero
            myRoot.AssemblyAngularVelocity = Vector3.zero

            OK_NukeForeignWelds(myChar)

            local target = TPlayers:FindFirstChild(targetName)
            if not target or not target.Parent then break end

            local tChar = target.Character
            local tRoot = tChar and tChar:FindFirstChild("HumanoidRootPart")
            local tHum  = tChar and tChar:FindFirstChildOfClass("Humanoid")

            if not (tRoot and tHum) or tHum.Health <= 0 then
                ownedRoot  = nil
                dragging   = false
                grabStartTime = 0
                checkStartTime = 0
                TRun.Heartbeat:Wait()
                continue
            end

            OK_NukeForeignWelds(tChar)

            local distToVictim = (myRoot.Position - tRoot.Position).Magnitude

            -- Resolve final anchor live every frame via the caller's getter
            -- so slider changes + your own movement are respected in real time.
            local baseAnchor
            if type(anchorGetter) == "function" then
                baseAnchor = anchorGetter() or tRoot.Position
            elseif typeof(anchorGetter) == "Vector3" then
                baseAnchor = anchorGetter
            else
                baseAnchor = tRoot.Position
            end

            if not dragging then
                local pcl  = OK_FindPCLD(target)
                local head = tChar:FindFirstChild("Head")

                -- Get in melee range without anchoring.
                if distToVictim > cfg.inRange then
                    pcall(function()
                        local dir = (tRoot.Position - myRoot.Position)
                        if dir.Magnitude > 0 then dir = dir.Unit end
                        myRoot.CFrame = CFrame.new(tRoot.Position - dir * 3, tRoot.Position)
                        myRoot.AssemblyLinearVelocity  = Vector3.zero
                        myRoot.AssemblyAngularVelocity = Vector3.zero
                    end)
                else
                    pcall(function()
                        myRoot.CFrame = tRoot.CFrame * CFrame.new(0, 0, 3)
                        myRoot.AssemblyLinearVelocity  = Vector3.zero
                        myRoot.AssemblyAngularVelocity = Vector3.zero
                    end)
                end

                local ackDeadline = tick() + cfg.ackTimeout
                local acked = false

                while tick() < ackDeadline do
                    -- Also budget the initial ack burst — otherwise 20 players
                    -- firing 4 remotes each per Heartbeat instantly saturates
                    -- the network and ping jumps to 400+.
                    local allow = RemoteBudget_Take(1)
                    if allow > 0 then
                        pcall(function()
                            TSetNetOwner:FireServer(tRoot, tRoot.CFrame)
                            if head then TSetNetOwner:FireServer(head, head.CFrame) end
                            if pcl then TSetNetOwner:FireServer(pcl, pcl.CFrame) end
                            TDestroyLine:FireServer(tRoot)
                        end)
                    end

                    if OK_WeOwnIt(tRoot) or OK_WeOwnIt(head) then
                        acked = true
                        break
                    end
                    TRun.Heartbeat:Wait()
                end

                if acked then
                    dragging = true
                    grabStartTime = 0
                    checkStartTime = tick()
                    ownedRoot = tRoot

                    if tHum.PlatformStand ~= true then tHum.PlatformStand = true end
                    if tHum.Sit          ~= true then tHum.Sit          = true end

                    tRoot.CFrame = CFrame.new(baseAnchor) * tRoot.CFrame.Rotation
                    tRoot.AssemblyLinearVelocity  = Vector3.zero
                    tRoot.AssemblyAngularVelocity = Vector3.zero

                    pcall(function() if TStruggle then TStruggle:FireServer(TLP) end end)
                else
                    if grabStartTime == 0 then grabStartTime = tick() end
                    if tick() - grabStartTime > cfg.grabWindow then
                        grabStartTime = 0
                    end
                end
            else
                -- ═══ LOCKED PHASE ═══
                if tHum.PlatformStand ~= true then tHum.PlatformStand = true end
                if tHum.Sit          ~= true then tHum.Sit          = true end

                -- LAYER 1: Ownership spam — budgeted across all targets and
                -- coalesced: we skip SetNetOwner if we already own the part.
                local now = tick()
                local spamInterval = cfg.getInterval()
                if not LastBurstTime[targetName] or now - LastBurstTime[targetName] >= spamInterval then
                    LastBurstTime[targetName] = now

                    local head = tChar:FindFirstChild("Head")
                    local pcl  = OK_FindPCLD(target)

                    -- Only spend a remote if we don't already own the part.
                    -- In multi-target scenarios most parts are already ours,
                    -- so this reduces traffic by 60-90% on subsequent ticks.
                    local wantRoot = not OK_WeOwnIt(tRoot)
                    local wantHead = head and not OK_WeOwnIt(head)
                    local wantPcl  = pcl  and not OK_WeOwnIt(pcl)

                    -- Head is the highest-priority part; HRP second. Take
                    -- from the global budget so multi-lock scenarios don't
                    -- saturate the outbound network.
                    if wantRoot then
                        local allow = RemoteBudget_Take(1)
                        if allow > 0 then
                            pcall(function()
                                TSetNetOwner:FireServer(tRoot, tRoot.CFrame)
                            end)
                        end
                    end
                    if wantHead then
                        local allow = RemoteBudget_Take(1)
                        if allow > 0 then
                            pcall(function()
                                TSetNetOwner:FireServer(head, head.CFrame)
                            end)
                        end
                    end
                    if wantPcl then
                        local allow = RemoteBudget_Take(1)
                        if allow > 0 then
                            pcall(function()
                                TSetNetOwner:FireServer(pcl, pcl.CFrame)
                            end)
                        end
                    end

                    -- Line-clear is cheap and important — keep it.
                    if StrongAntiLine then
                        pcall(function()
                            TDestroyLine:FireServer(tRoot)
                            if head then TDestroyLine:FireServer(head) end
                        end)
                    end
                end

                -- Move victim via their own root CFrame (server-authoritative)
                -- anchorPos already includes XYZ offset applied by the caller.
                pcall(function()
                    tRoot.CFrame = CFrame.new(baseAnchor) * tRoot.CFrame.Rotation
                    tRoot.AssemblyLinearVelocity  = Vector3.zero
                    tRoot.AssemblyAngularVelocity = Vector3.zero
                end)

                -- LAYER 4: Ragdoll reset
                if StrongRagdollReset then
                    pcall(function()
                        local rr = TCharEvents and TCharEvents:FindFirstChild("RagdollRemote")
                        if rr then rr:FireServer(tRoot, 0) end
                    end)
                end

                -- LAYER 5: Line clear handled above alongside ownership spam

                if cfg.escapeRecover and checkStartTime > 0 and tick() - checkStartTime > 0.30 then
                    if (tRoot.Position - baseAnchor).Magnitude > cfg.maxDist then
                        dragging = false
                        grabStartTime = 0
                        checkStartTime = 0
                        ownedRoot = nil
                    else
                        checkStartTime = tick()
                    end
                end
            end

            TRun.Heartbeat:Wait()
        end

        -- Release victim's state
        pcall(function()
            local target = TPlayers:FindFirstChild(targetName)
            local tChar = target and target.Character
            local tHum = tChar and tChar:FindFirstChildOfClass("Humanoid")
            if tHum then
                tHum.PlatformStand = false
                tHum.Sit = false
            end
            local tRoot = tChar and tChar:FindFirstChild("HumanoidRootPart")
            if tRoot then
                tRoot.Anchored = false
                tRoot.AssemblyLinearVelocity  = Vector3.zero
                tRoot.AssemblyAngularVelocity = Vector3.zero
            end
        end)

        pcall(function()
            if myRoot and myRoot.Parent then
                myRoot.Anchored = savedAnchored or false
                myRoot.CFrame   = savedPos
                myRoot.AssemblyLinearVelocity  = Vector3.zero
                myRoot.AssemblyAngularVelocity = Vector3.zero
            end
        end)
    end

    local TRemoveGucci_VOID_DIST = 500000

    local function RG_FindSeatables(plr)
        local seatables = {}
        local seen = {}

        local function addSeatable(s)
            if s and s:IsA("BasePart") and not seen[s] then
                seen[s] = true
                table.insert(seatables, s)
            end
        end

        local folder = TWS:FindFirstChild(plr.Name .. "SpawnedInToys")
        if folder then
            for _, obj in ipairs(folder:GetDescendants()) do
                if obj:IsA("VehicleSeat") or obj:IsA("Seat") then
                    addSeatable(obj)
                end
            end

            for _, obj in ipairs(folder:GetChildren()) do
                if obj:IsA("Model")
                    and not obj:FindFirstChildWhichIsA("VehicleSeat", true)
                    and not obj:FindFirstChildWhichIsA("Seat", true) then
                    local root = obj.PrimaryPart
                        or obj:FindFirstChild("SoundPart", true)
                        or obj:FindFirstChildWhichIsA("BasePart", true)
                    addSeatable(root)
                end
            end
        end

        -- 2) Any Seat the target's Humanoid is currently sitting on.
        local tChar = plr.Character
        local tHum = tChar and tChar:FindFirstChildOfClass("Humanoid")
        if tHum and tHum.SeatPart then
            addSeatable(tHum.SeatPart)
        end

        -- 3) Anywhere in workspace where the model name matches CreatureBlobman
        -- and the model's creator is the target (SpawnedInToys path or PartOwner).
        for _, obj in ipairs(TWS:GetDescendants()) do
            if obj:IsA("VehicleSeat") or obj:IsA("Seat") then
                local owner = obj:FindFirstChild("PartOwner", true)
                if owner and owner:IsA("StringValue") and owner.Value == plr.Name then
                    addSeatable(obj)
                end
            end
        end

        return seatables
    end

    -- Fling a seatable far into the void. Needs net-owner first or the
    -- server will just snap it back.
    local function RG_VoidSeatable(seat)
        if not seat or not seat.Parent then return end

        -- Grab ownership once. Retry a couple times if the server resists.
        local owns = false
        for _ = 1, 3 do
            if RG_TakeOwnership(seat) then owns = true break end
        end
        if not owns then return end

        -- Yank it hard: direct CFrame + velocity is the most reliable
        -- combination. The server's FallenPartsDestroyHeight will clean it up.
        pcall(function()
            seat.CFrame = CFrame.new(
                seat.Position.X,
                -TRemoveGucci_VOID_DIST,
                seat.Position.Z
            )
            seat.AssemblyLinearVelocity  = Vector3.new(0, -TRemoveGucci_VOID_DIST, 0)
            seat.AssemblyAngularVelocity = Vector3.zero
        end)

        -- Belt-and-suspenders: keep pushing it down for a few frames in case
        -- the physics solver tries to fight us.
        local stopAt = tick() + 0.35
        while tick() < stopAt and seat.Parent do
            pcall(function()
                seat.CFrame = CFrame.new(
                    seat.Position.X,
                    seat.Position.Y - 50000,
                    seat.Position.Z
                )
                seat.AssemblyLinearVelocity = Vector3.new(0, -9e7, 0)
            end)
            TRun.Heartbeat:Wait()
        end
    end

    local function TRemoveGucci(targetName)
        local target = TPlayers:FindFirstChild(targetName)
        if not target then return end

        local myChar = TLP.Character
        local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
        local myHum  = myChar and myChar:FindFirstChildOfClass("Humanoid")
        if not (myRoot and myHum) then return end

        -- Save where we were so we can teleport back.
        local SafeSpot = myRoot.CFrame

        task.spawn(function()
            -- ── 1. Fling every seatable we can find into the void. ──
            local seatables = RG_FindSeatables(target)
            for _, seat in ipairs(seatables) do
                task.spawn(function()
                    RG_VoidSeatable(seat)
                end)
                -- Small stagger so we don't saturate the network.
                task.wait(0.02)
            end

            -- ── 2. Grab the target: teleport onto them and take ownership. ──
            local tChar = target.Character
            local tRoot = tChar and tChar:FindFirstChild("HumanoidRootPart")
            local tHum  = tChar and tChar:FindFirstChildOfClass("Humanoid")
            if not (tRoot and tHum) then
                -- Nothing to grab — just go home.
                pcall(function()
                    myRoot.CFrame = SafeSpot
                    myRoot.AssemblyLinearVelocity  = Vector3.zero
                    myRoot.AssemblyAngularVelocity = Vector3.zero
                end)
                return
            end

            -- Teleport behind them and grab their HRP + Head.
            pcall(function()
                myRoot.CFrame = tRoot.CFrame * CFrame.new(0, 0, 3)
                myRoot.AssemblyLinearVelocity  = Vector3.zero
                myRoot.AssemblyAngularVelocity = Vector3.zero
            end)

            local grabDeadline = tick() + TRemoveGucci_GRAB_TIME
            while tick() < grabDeadline do
                if not target.Parent or not target.Character then break end
                local tCharNow = target.Character
                local tRootNow = tCharNow and tCharNow:FindFirstChild("HumanoidRootPart")
                local tHeadNow = tCharNow and tCharNow:FindFirstChild("Head")
                if not tRootNow then break end

                pcall(function()
                    TSetNetOwner:FireServer(tRootNow, tRootNow.CFrame)
                    if tHeadNow then
                        TSetNetOwner:FireServer(tHeadNow, tHeadNow.CFrame)
                    end
                    if StrongAntiLine then
                        TDestroyLine:FireServer(tRootNow)
                        if tHeadNow then TDestroyLine:FireServer(tHeadNow) end
                    end
                end)

                -- Pin them so they can't wriggle out.
                if OK_WeOwnIt(tRootNow) then
                    pcall(function()
                        tRootNow.AssemblyLinearVelocity  = Vector3.zero
                        tRootNow.AssemblyAngularVelocity = Vector3.zero
                    end)
                    if tHum and tHum.PlatformStand ~= true then
                        tHum.PlatformStand = true
                    end
                end

                TRun.Heartbeat:Wait()
            end

            -- ── 3. Release the target's state so they aren't frozen. ──
            pcall(function()
                local tCharNow = target.Character
                local tHumNow = tCharNow and tCharNow:FindFirstChildOfClass("Humanoid")
                if tHumNow then
                    tHumNow.PlatformStand = false
                    tHumNow.Sit = false
                end
            end)

            -- ── 4. Teleport back to where we were. ──
            pcall(function()
                myRoot.CFrame = SafeSpot
                myRoot.AssemblyLinearVelocity  = Vector3.zero
                myRoot.AssemblyAngularVelocity = Vector3.zero
            end)
        end)
    end

--/(method dispatch)
    -- NOTE: assigned (not re-declared) so it binds to the forward-declared
    -- local above. Do NOT use `local function` here or it creates a new
    -- local that shadows the outer one and leaves the Apply buttons calling nil.
    TStartOwnershipKickForPlayer = function(plr, anchorGetter)
        if not plr or not plr.Parent then return end
        local name = plr.Name
        -- If a kick is already running, just refresh the anchor getter so the
        -- new offset values are picked up live instead of being ignored.
        if OwnershipKickThread[name] then
            OwnershipKickRefs[name] = anchorGetter
            return
        end
        OwnershipKickActive[name] = true
        OwnershipKickRefs[name]   = anchorGetter
        -- Flag as a kick target so PlayerRemoving fires the kick notification.
        TKickTargets[plr.UserId] = true
        OwnershipKickThread[name] = task.spawn(function()
            RunOwnershipKick(name, function()
                return OwnershipKickActive[name] == true
            end, function()
                local g = OwnershipKickRefs[name]
                if type(g) == "function" then return g() end
                if typeof(g) == "Vector3" then return g end
                return nil
            end)
            OwnershipKickActive[name] = nil
            OwnershipKickThread[name] = nil
            OwnershipKickRefs[name]   = nil
            CleanupPalletFor(name)
            CleanupBodiesFor(name)
        end)
        TRegisterThread(OwnershipKickThread[name])
    end

--/(loop snowball — spam, no spawn cooldown)
    local function RunLoopSnowball()
        local inv = TWS:FindFirstChild(TLP.Name .. "SpawnedInToys")
        while LoopSnowballActive do
            inv = TWS:FindFirstChild(TLP.Name .. "SpawnedInToys") or inv
            for _, plr in ipairs(TGetSelectedList()) do
                local tChar = plr.Character
                if tChar then
                    local torso = tChar:FindFirstChild("HumanoidRootPart") or tChar:FindFirstChild("Torso")
                    if torso then
                        -- Budget spawn calls too — InvokeServer is a yieldable
                        -- remote function so each call costs ~2 network round
                        -- trips. Spraying 3× per target per frame = massive ping.
                        local allow = RemoteBudget_Take(1)
                        if allow > 0 then
                            local offset = Vector3.new(
                                math.random(-5, 5) / 10,
                                math.random(-5, 5) / 10,
                                math.random(-5, 5) / 10
                            )
                            task.spawn(function()
                                pcall(function()
                                    TSpawnToy:InvokeServer("BallSnowball", torso.CFrame * CFrame.new(offset), Vector3.zero)
                                end)
                            end)
                        end
                        if inv then
                            for _, sb in ipairs(inv:GetChildren()) do
                                if sb.Name == "BallSnowball" then
                                    local part = sb:IsA("BasePart") and sb
                                        or (sb.PrimaryPart or sb:FindFirstChildWhichIsA("BasePart"))
                                    if part then
                                        pcall(function()
                                            part.CFrame = torso.CFrame
                                            part.AssemblyLinearVelocity = Vector3.zero
                                            part.AssemblyAngularVelocity = Vector3.zero
                                        end)
                                    end
                                end
                            end
                        end
                    end
                end
            end
            TRun.Heartbeat:Wait()
        end
    end

--/(loop ragdoll — Genesis v2 style: spam RagdollRemote with no spawn cooldown)
    local function CleanupRagdoll()
        RagdollPallet = nil
        RagdollSoundPart = nil
    end

    -- Genesis v2 approach: no pallet needed. Hammer RagdollRemote against
    -- both HRP and Head, reclaim net-owner every frame, and spam ragdoll
    -- state so anti-ragdoll recoveries can't get up.
    local function RunLoopRagdoll()
        while LoopRagdollActive do
            for _, plr in ipairs(TGetSelectedList()) do
                local tChar = plr.Character
                local tRoot = tChar and tChar:FindFirstChild("HumanoidRootPart")
                local tHead = tChar and tChar:FindFirstChild("Head")
                local tHum  = tChar and tChar:FindFirstChildOfClass("Humanoid")
                if tRoot and tHum and tHum.Health > 0 then
                    -- Budget every remote — multiple targets × 6 remotes each
                    -- at 60 Hz = ~1400 remotes/sec per target, and that's the
                    -- single biggest ping killer here.
                    local allow = RemoteBudget_Take(1)
                    if allow > 0 then
                        pcall(function()
                            TSetNetOwner:FireServer(tRoot, tRoot.CFrame)
                            if tHead then TSetNetOwner:FireServer(tHead, tHead.CFrame) end
                        end)
                    end
                    if TCharEvents and TCharEvents:FindFirstChild("RagdollRemote") then
                        pcall(function()
                            -- Only one ragdoll pair (1,0) per frame per target —
                            -- the server reapplies state fast enough that
                            -- spraying more only adds latency.
                            TCharEvents.RagdollRemote:FireServer(tRoot, 1)
                            TCharEvents.RagdollRemote:FireServer(tRoot, 0)
                        end)
                    end
                end
            end
            TRun.Heartbeat:Wait()
        end
        CleanupRagdoll()
    end
    function TStartLoopRagdoll()
        if LoopRagdollThread then task.cancel(LoopRagdollThread) end
        LoopRagdollActive = true
        LoopRagdollThread = task.spawn(RunLoopRagdoll)
        TRegisterThread(LoopRagdollThread)
        TAddCleanup(function() TStopLoopRagdoll() end)
    end

    function TStopLoopRagdoll()
        LoopRagdollActive = false
        if LoopRagdollThread then
            pcall(function() task.cancel(LoopRagdollThread) end)
            LoopRagdollThread = nil
        end
        if RagdollPallet and RagdollPallet.Parent then
            pcall(function() TDestroyToy:FireServer(RagdollPallet) end)
        end
        CleanupRagdoll()
    end

                            if ragdolled and ragdolled.Value and not onBlobSeat() then
                                nrdDisableLimbs(char)
                            end
                        end)
                        task.wait()
                    end
                end)
                pcall(function() hrp.Anchored = false end)
                if nrd._token == token then nrd.proc = false end
                if P.Character == char and not characterIsGrabbed(char) then
                    task.spawn(clearGrabLines)
                end
            end)
        end

            if not antiGrabAlive(nrd, token, char) then return end

            beginEscape()

    function checkgrab(obj)
        if obj:FindFirstAncestor('Plots') or obj:FindFirstAncestor('Slots') or obj:FindFirstAncestor('Map') then
            return
        end
        local prt = obj:FindFirstChild('SoundPart') or obj:FindFirstChild('HumanoidRootPart')
            or obj:FindFirstChild('Main') or obj:FindFirstChild('Balloon')
        if obj:FindFirstChild('Humanoid') then
            if not obj.Humanoid.SeatPart then
                if prt and prt.ReceiveAge ~= 0 then
                    return false
                end
            elseif obj.Humanoid.SeatPart.Parent then
                return checkgrab(obj.Humanoid.SeatPart.Parent)
            end
        elseif prt and prt.ReceiveAge ~= 0 then
            return false
        end
        return true
    end

    function cheekyAntiGrab()
        local model
        while ACGE do
            local char = P.Character
            if char and char:FindFirstChild('Humanoid') then
                for _, track in ipairs(char.Humanoid:GetPlayingAnimationTracks()) do
                    if track.Animation.AnimationId == 'rbxassetid://7047322890' then
                        track:Stop()

                end
            end

            local toyFolder = WS:FindFirstChild(P.Name .. 'SpawnedInToys')
            if toyFolder then
                model = toyFolder:FindFirstChild('FoodMayonnaise')
            end

            if char and not checkgrab(char) then
                local hum = char:FindFirstChild('Humanoid')
                local ragdolled = hum and hum:FindFirstChild('Ragdolled') and hum.Ragdolled.Value

                if ragdolled or not model then
                    pcall(function() StruggleEvent:FireServer(P) end)
                    if hum then
                        for _, track in ipairs(hum:GetPlayingAnimationTracks()) do
                            if track.Animation.AnimationId == 'rbxassetid://7047322890' then
                                track:Stop()
                            end
                        end
                    end
                else
                    if model and model:FindFirstChild('HoldPart') then
                        local hold = model.HoldPart
                        if hold:FindFirstChild('HoldItemRemoteFunction') then
                            task.spawn(function()
                                pcall(function()
                                    hold.HoldItemRemoteFunction:InvokeServer(model, char)
                                end)
                            end)
                            pcall(function() DestroyToy:FireServer(model) end)
                            if hum then
                                hum:SetStateEnabled(Enum.HumanoidStateType.Jumping, true)
                                hum.AutoRotate = true
                                hum.Sit = false
                            end
                            pcall(function()
                                ContextActionService:UnbindAction('Escape')
                                ContextActionService:UnbindAction('JumpRemover')

            if hum then

                pcall(function() hum.PlatformStand = false end)

    -- ══════════════════════════════════════════════════════════════
    -- UNIVERSAL SHIELD — constraint sweep + network-ownership reclaim
    -- Runs every Heartbeat when Anti Grab is enabled (both modes).
    -- Destroys any Weld / WeldConstraint / RopeConstraint /
    -- SpringConstraint that originates from another player's assembly.
    -- Also re-claims SetNetworkOwner on HRP every 60 frames.
    -- ══════════════════════════════════════════════════════════════

    local HOSTILE_CONSTRAINT_TYPES = {
        WeldConstraint = true,
        Weld = true,
        RopeConstraint = true,
        SpringConstraint = true,
        RodConstraint = true,
        BallSocketConstraint = true,
        HingeConstraint = true,
        AlignPosition = true,
        AlignOrientation = true,
        LinearVelocity = true,
        AngularVelocity = true,
        BodyVelocity = true,
        BodyAngularVelocity = true,
        BodyPosition = true,
        BodyGyro = true,
        BodyForce = true,
        VectorForce = true,
        Torque = true,
    }

    local universalShieldConn = nil
    local universalShieldFrame = 0

    local function isPartFromOtherPlayer(part)
        if not part or not part:IsA("BasePart") then return false end
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= P and p.Character and part:IsDescendantOf(p.Character) then
                return true
            end
        end
        return false
    end

    local function sweepConstraints(character)
        for _, desc in ipairs(character:GetDescendants()) do
            if HOSTILE_CONSTRAINT_TYPES[desc.ClassName] then
                local shouldNuke = false
                pcall(function()
                    if desc:IsA("WeldConstraint") or desc:IsA("Weld") then
                        shouldNuke = isPartFromOtherPlayer(desc.Part0)
                                  or isPartFromOtherPlayer(desc.Part1)
                    elseif desc.ClassName == "RopeConstraint"
                        or desc.ClassName == "SpringConstraint"
                        or desc.ClassName == "RodConstraint"
                        or desc.ClassName == "BallSocketConstraint"
                        or desc.ClassName == "HingeConstraint" then
                        local a0 = desc.Attachment0 and desc.Attachment0.Parent
                        local a1 = desc.Attachment1 and desc.Attachment1.Parent
                        shouldNuke = isPartFromOtherPlayer(a0)
                                  or isPartFromOtherPlayer(a1)
                    else
                        -- Movers: nuke in NoRagdoll when parent is external.
                        if antiGrabMode == "No Ragdoll" then
                            shouldNuke = not desc.Parent:IsDescendantOf(character)
                        else
                            shouldNuke = isPartFromOtherPlayer(desc.Parent)
                        end
                    end
                end)
                if shouldNuke then
                    pcall(function() desc:Destroy() end)
                end
            end
        end
    end

    local function startUniversalShield()
        if universalShieldConn then return end
        universalShieldConn = AntiGrabRunService.Heartbeat:Connect(function()
            if not antiGrabEnabled then return end
            local character = P.Character
            if not character then return end

            -- Constraint sweep every frame.
            sweepConstraints(character)

            -- NoRagdoll extras: keep state from flipping, clamp fling velocity.
            if antiGrabMode == "No Ragdoll" then
                local hum = character:FindFirstChildOfClass("Humanoid")
                local hrp = character:FindFirstChild("HumanoidRootPart")
                if hum then
                    if hum.PlatformStand then hum.PlatformStand = false end
                    -- Do not cancel a legitimate CreatureBlobman seat.
                    -- The old No Ragdoll heartbeat forced Humanoid.Sit = false
                    -- every frame, which made it impossible to stay seated.
                    if hum.Sit and not onBlobSeat() then
                        hum.Sit = false
                    end
                end
                if hrp and hrp.AssemblyLinearVelocity.Magnitude > 220 then
                    pcall(function()
                        hrp.AssemblyLinearVelocity = Vector3.zero
                    end)
                end
            end

            -- Network ownership reclaim every 60 frames.
            universalShieldFrame = universalShieldFrame + 1
            if universalShieldFrame >= 60 then
                universalShieldFrame = 0
                local hrp = character:FindFirstChild("HumanoidRootPart")
                if hrp then
                    pcall(function() SetNetOwner:FireServer(hrp, hrp.CFrame) end)
                end
            end
        end)
    end

    local function stopUniversalShield()
        if universalShieldConn then
            pcall(function() universalShieldConn:Disconnect() end)
            universalShieldConn = nil
        end
        universalShieldFrame = 0
    end

    local function safeGetDescendants(instance)
        if typeof(instance) ~= "Instance" then return {} end
        local ok, result = pcall(function() return instance:GetDescendants() end)
        if ok and type(result) == "table" then return result end
        return {}
    end

    local function watchGucciToy(toy)
        if not toy then return end
        if gucciDestroyWatch then
            pcall(function() gucciDestroyWatch:Disconnect() end)
        end
        gucciLastKnownPlayer = nil
        gucciDestroyWatch = toy.AncestryChanged:Connect(function(_, parent)
            if parent ~= nil or not antiGucciEnabled or gucciToy ~= toy then return end
            local remover = gucciLastKnownPlayer
            pcall(function()
                local seat = toy:FindFirstChild("VehicleSeat", true)
                if not remover and seat and seat.Occupant and seat.Occupant.Parent then
                    local player = Players:GetPlayerFromCharacter(seat.Occupant.Parent)
                    if player and player ~= P then remover = player end
                end
                if not remover then
                    for _, obj in ipairs(safeGetDescendants(toy)) do
                        if obj.Name == "PartOwner" and obj:IsA("StringValue") then
                            local player = Players:FindFirstChild(obj.Value)
                            if player and player ~= P then
                                remover = player
                                break
                            end
                        end
                    end
                end
            end)
            if remover then
                _G.__EchoNotify("You're Gucci Has been removed by " .. remover.DisplayName .. " (" .. remover.Name .. ")", "gucci_destroyed", 5, "GucciDestroyed")
            else
                _G.__EchoNotify("You're Gucci Has been removed", "gucci_destroyed", 5, "GucciDestroyed")
            end
            gucciToy = nil
            gucciInSeat = false
        end)

        if not toy or not toy.Parent then return false end
        local descendants = {}
        local ok = pcall(function()
            descendants = safeGetDescendants(toy)
        end)
        if not ok then return false end
        for _, part in ipairs(descendants) do
            if part:IsA("BasePart") then
                part.CanCollide = false
                part.CanTouch = false
                part.CanQuery = false
                part.AssemblyAngularVelocity = Vector3.zero
                if hide then
                    part.Transparency = 1
                end
            end
        end
        return true
    end

    local function runGucciToy(Session, toyName, hide)
        if gucciRunning then return end
        gucciRunning = true
        local toyAliveSince = 0
        local guccion = false
        local savedCFrame = nil
        local savedVelocity = nil
        while antiGucciEnabled and Session == antiGucciSession and _G.__EchoGucciRuntimeToken == gucciRuntimeToken do
            local hum, root = getCharParts()
            gucciHum, gucciRoot = hum, root
            if not hum or not root or not root.Parent then
                task.wait(0.1)
                continue
            end
            if not savedCFrame or not hum.SeatPart then
                savedCFrame = root.CFrame
                savedVelocity = root.AssemblyLinearVelocity
            end
            local folder = WS:FindFirstChild(MI_CARPETA)
            local toy = gucciToy
            if not toy or not toy.Parent then
                toy = folder and folder:FindFirstChild(toyName)
                gucciToy = toy
                gucciInSeat = false
                guccion = false
                toyAliveSince = toy and tick() or 0
                if toy then watchGucciToy(toy) end
            end
            if not toy then
                if tick() - toyAliveSince > 0.15 then
                    spawnSelectedGucci(toyName, root)
                    toyAliveSince = tick()
                end
                task.wait(0.08)
                continue
            end
            local seat = toy:FindFirstChild("VehicleSeat", true) or toy:FindFirstChildWhichIsA("VehicleSeat", true)
            local toyRoot = toy:FindFirstChild("HumanoidRootPart", true) or toy:FindFirstChild("SoundPart", true)
            if not seat or not toyRoot or not seat.Parent or not toyRoot.Parent then
                if tick() - toyAliveSince >= 0.75 then
                    destroyGucciToy()
                    gucciInSeat = false
                    guccion = false
                    toyAliveSince = 0
                end
                task.wait(0.05)
                continue
            end
            if gucciSeatFor ~= seat then
                if gucciSeatConn then pcall(function() gucciSeatConn:Disconnect() end) end
                gucciSeatFor = seat
                gucciSeatConn = seat:GetPropertyChangedSignal("Occupant"):Connect(function()
                    if not antiGucciEnabled or Session ~= antiGucciSession then return end
                    local occupant = seat.Occupant
                    if occupant and occupant ~= hum then
                        destroyGucciToy()
                        gucciInSeat = false
                        guccion = false
                    end
                end)
            end
            if seat.Occupant and seat.Occupant ~= hum then
                destroyGucciToy()
                gucciInSeat = false
                guccion = false
                task.wait(0.1)
                continue
            end
            for _, obj in ipairs(safeGetDescendants(toy)) do
                if obj:IsA("BasePart") then
                    obj.CanCollide = false
                    obj.CanTouch = false
                    obj.CanQuery = false
                    if hide then obj.Transparency = 1 end
                end
            end
            pcall(function() seat.Parent = nil; seat.Parent = toy end)
            if not hum.SeatPart then
                hum.Sit = false
                if not hide then
                    seat.CFrame = ALTO
                end
                root.CFrame = seat.CFrame + Vector3.new(0, 2, 0)
                seat:Sit(hum)
                if RagdollRemote then RagdollRemote:FireServer(root, 0.016) end
                task.wait()
            end
            if seat.Occupant == hum or hum.SeatPart == seat then
                gucciInSeat = true
                guccion = true
                if not savedCFrame then savedCFrame = root.CFrame end
                if RagdollRemote then pcall(function() RagdollRemote:FireServer(root, 0.016) end) end
                if toyRoot:IsA("BasePart") then
                    pcall(function()
                        toyRoot.AssemblyLinearVelocity = UP_1E15
                        toyRoot.AssemblyAngularVelocity = Vector3.zero
                    end)
                end
            else
                gucciInSeat = false
                guccion = false
            end
            if guccion then
                pcall(function()
                    root.CFrame = savedCFrame
                    if savedVelocity then root.AssemblyLinearVelocity = savedVelocity end
                end)
            end
            if not toy.Parent then
                gucciToy = nil
                gucciInSeat = false
                guccion = false
            end
            task.wait()
        end
        gucciRunning = false
    end

                end)

function makeHighlight(store, key, adornee, name, fill, fillTransparency, outline, outlineTransparency)

    local h = store[key]

    if not h or not h.Parent then

        h = Instance.new("Highlight")

        h.Name = name

        h.Parent = ESPFolder

        store[key] = h

    end

    h.Adornee = adornee

    h.FillColor = fill

    h.FillTransparency = fillTransparency

    h.OutlineColor = outline

    h.OutlineTransparency = outlineTransparency

    h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop

    h.Enabled = true

    return h

function clearPlayerESP(player)

    local h = PlayerHighlights[player]

    if h then pcall(function() h:Destroy() end); PlayerHighlights[player] = nil end

end

function updatePlayerESP(player)

    if player == LocalPlayer or not VisualState.PlayerESP then clearPlayerESP(player); return end

    local character = player.Character

    if not character or not character.Parent then clearPlayerESP(player); return end

    local h = PlayerHighlights[player]

    if not h or not h.Parent then

        h = Instance.new("Highlight")

        h.Name = "Echo_PlayerESP"

        h.Parent = character

        PlayerHighlights[player] = h

    elseif h.Parent ~= character then h.Parent = character end

    h.Adornee = character

    h.FillColor = VisualState.PlayerFillColor

    h.FillTransparency = 0.5

    h.OutlineColor = VisualState.PlayerOutlineColor

    h.OutlineTransparency = 0

    h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop

    h.Enabled = true

end

function refreshPlayerESP()

    if not VisualState.PlayerESP then destroyMap(PlayerHighlights); return end

    for _, player in ipairs(Players:GetPlayers()) do updatePlayerESP(player) end

end

-- Tracks default game overhead tags we hid so we can restore them.
local NameHiddenDefaults = {}   -- [player] = { [instance] = true }

-- Hide every non-Echo GUI on the player's character (default name tag +
-- any HealthGui / overhead UI the game may attach anywhere on the model).
local function hideDefaultNameTags(player)
    local char = player and player.Character
    if not char then return end

    NameHiddenDefaults[player] = NameHiddenDefaults[player] or {}
    local tracked = NameHiddenDefaults[player]

    -- Common name-tag root names used by most Roblox games.
    local IGNORE_NAMES = {
        HumanoidRootPart = true,
        Head = true,
        Torso = true,
        ["Left Arm"] = true,
        ["Right Arm"] = true,
        ["Left Leg"] = true,
        ["Right Leg"] = true,
        UpperTorso = true,
        LowerTorso = true,
        LeftUpperArm = true,
        LeftLowerArm = true,
        LeftHand = true,
        RightUpperArm = true,
        RightLowerArm = true,
        RightHand = true,
        LeftUpperLeg = true,
        LeftLowerLeg = true,
        LeftFoot = true,
        RightUpperLeg = true,
        RightLowerLeg = true,
        RightFoot = true,
    }

    for _, obj in ipairs(char:GetDescendants()) do
        -- Never touch our own ESP label.
        if obj.Name ~= "Echo_NameESP" then
            local hide = false

            -- Any BillboardGui / SurfaceGui / ScreenGui that isn't ours.
            if obj:IsA("BillboardGui") or obj:IsA("SurfaceGui") or obj:IsA("ScreenGui") then
                -- Skip our own ESP (its name is Echo_NameESP).
                if not (obj.Name == "Echo_NameESP" or (obj.Parent and obj.Parent.Name == "Echo_NameESP")) then
                    hide = true
                end
            -- Loose TextLabels directly on a body part (some games do this).
            elseif obj:IsA("TextLabel") then
                if IGNORE_NAMES[obj.Name] == nil and obj.Parent and IGNORE_NAMES[obj.Parent.Name] then
                    hide = true
                end
            end

            if hide then
                pcall(function()
                    if obj:IsA("BillboardGui") or obj:IsA("SurfaceGui") or obj:IsA("ScreenGui") then
                        if obj.Enabled then tracked[obj] = true end
                        obj.Enabled = false
                    elseif obj:IsA("TextLabel") then
                        if obj.Visible then tracked[obj] = true end
                        obj.Visible = false
                    end
                end)
            end
        end
    end
end

-- Restore any previously hidden default tags for this player.
local function restoreDefaultNameTags(player)
    local tracked = NameHiddenDefaults[player]
    if not tracked then return end
    for obj in pairs(tracked) do
        pcall(function()
            if obj and obj.Parent then
                if obj:IsA("BillboardGui") or obj:IsA("SurfaceGui") or obj:IsA("ScreenGui") then
                    obj.Enabled = true
                elseif obj:IsA("TextLabel") then
                    obj.Visible = true
                end
            end
        end)
    end
    NameHiddenDefaults[player] = nil
end

-- Persistent watcher: re-hide any tag the game re-enables while ESP is on.
local NameTagWatchConns = {}   -- [player] = { conn1, conn2, ... }
local function watchDefaultNameTags(player)
    local char = player and player.Character
    if not char then return end
    -- Disconnect existing watchers for this player.
    if NameTagWatchConns[player] then
        for _, c in ipairs(NameTagWatchConns[player]) do pcall(function() c:Disconnect() end) end
    end
    NameTagWatchConns[player] = {}

    local function watchOne(obj)
        if not obj or obj.Name == "Echo_NameESP" then return end
        if not (obj:IsA("BillboardGui") or obj:IsA("SurfaceGui") or obj:IsA("ScreenGui") or obj:IsA("TextLabel")) then return end
        local conn
        if obj:IsA("TextLabel") then
            conn = obj:GetPropertyChangedSignal("Visible"):Connect(function()
                if VisualState.NameESP and obj.Visible then
                    obj.Visible = false
                end
            end)
        else
            conn = obj:GetPropertyChangedSignal("Enabled"):Connect(function()
                if VisualState.NameESP and obj.Enabled then
                    obj.Enabled = false
                end
            end)
        end
        table.insert(NameTagWatchConns[player], conn)
    end

    for _, obj in ipairs(char:GetDescendants()) do
        watchOne(obj)
    end
    -- Also catch anything added later (respawn, tool equipped, etc.).
    table.insert(NameTagWatchConns[player], char.DescendantAdded:Connect(function(obj)
        task.defer(function()
            if VisualState.NameESP and obj and obj.Parent then
                watchOne(obj)
                if obj.Name ~= "Echo_NameESP" and (obj:IsA("BillboardGui") or obj:IsA("SurfaceGui") or obj:IsA("ScreenGui")) then
                    obj.Enabled = false
                elseif obj:IsA("TextLabel") and obj.Name ~= "Echo_NameESP" then
                    obj.Visible = false
                end
            end
        end)
    end))
end

local function unwatchDefaultNameTags(player)
    local conns = NameTagWatchConns[player]
    if conns then
        for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
        NameTagWatchConns[player] = nil
    end
end
local function getNameESPDistance(player)
    local character = player and player.Character
    local targetRoot = character and character:FindFirstChild("HumanoidRootPart")
    local localCharacter = LocalPlayer and LocalPlayer.Character
    local localRoot = localCharacter and localCharacter:FindFirstChild("HumanoidRootPart")
    if not targetRoot or not localRoot then return "0m" end
    return string.format("%.0fm", (targetRoot.Position - localRoot.Position).Magnitude)
end
local function getPlayerIcon(player)
    if not player then return "" end
    local cached = NameIconCache[player.UserId]
    if cached then return cached end
    local ok, content = pcall(function()
        return Players:GetUserThumbnailAsync(player.UserId, Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size100x100)
    end)
    if ok and content and content ~= "" then
        NameIconCache[player.UserId] = content
        return content
    end
    return ""
end
function clearNameESP(player)
    local state = NameStates[player]
    if state then
        if state.gui then pcall(function() state.gui:Destroy() end) end
        NameStates[player] = nil
    end
    -- Bring back the game's default name tag when Echo ESP is off.
    unwatchDefaultNameTags(player)
    restoreDefaultNameTags(player)
end
local function updateNameESPText(player, state)
    if not state or not state.label then return end
    state.label.Text = player.DisplayName .. "  " .. getNameESPDistance(player) .. "\n(" .. player.Name .. ")"
end
function updateNameESP(player)
    if player == LocalPlayer or not VisualState.NameESP then
        clearNameESP(player)
        return
    end
    local character = player.Character
    local head = character and character:FindFirstChild("Head")
    if not character or not character.Parent or not head then
        clearNameESP(player)
        return
    end
    local state = NameStates[player]
    local gui = state and state.gui
    if not gui or not gui.Parent or gui.Adornee ~= head then
        if gui then pcall(function() gui:Destroy() end) end
        gui = Instance.new("BillboardGui")
        gui.Name = "Echo_NameESP"
        gui.Adornee = head
        gui.Size = UDim2.fromOffset(220, 68)
        gui.StudsOffset = Vector3.new(0, 3.45, 0)
        gui.AlwaysOnTop = true
        gui.MaxDistance = 750
        gui.LightInfluence = 0
        gui.ResetOnSpawn = false
        gui.Parent = head
        local icon = Instance.new("ImageLabel")
        icon.Name = "PlayerIcon"
        icon.BackgroundTransparency = 1
        icon.AnchorPoint = Vector2.new(0.5, 0)
        icon.Position = UDim2.new(0.5, 0, 0, 0)
        icon.Size = UDim2.fromOffset(26, 26)
        icon.ScaleType = Enum.ScaleType.Fit
        icon.ImageTransparency = 0
        icon.Image = getPlayerIcon(player)
        icon.Parent = gui
        local label = Instance.new("TextLabel")
        label.Name = "PlayerName"
        label.BackgroundTransparency = 1
        label.AnchorPoint = Vector2.new(0.5, 0)
        label.Position = UDim2.new(0.5, 0, 0, 27)
        label.Size = UDim2.new(1, 0, 0, 38)
        label.Font = Enum.Font.Gotham
        label.TextSize = 14
        label.TextWrapped = false
        label.TextTruncate = Enum.TextTruncate.AtEnd
        label.TextXAlignment = Enum.TextXAlignment.Center
        label.TextYAlignment = Enum.TextYAlignment.Top
        label.TextStrokeTransparency = 0.5
        label.TextStrokeColor3 = VisualState.NameOutlineColor
        label.TextColor3 = VisualState.NameColor
        label.Parent = gui
        state = {gui = gui, icon = icon, label = label, head = head}
        NameStates[player] = state
    end
    state.head = head
    gui.Adornee = head
    gui.Enabled = true
    if state.icon then
        local cachedIcon = NameIconCache[player.UserId]
        if cachedIcon then state.icon.Image = cachedIcon end
    end
    if state.label then
        state.label.TextColor3 = VisualState.NameColor
        state.label.TextStrokeColor3 = VisualState.NameOutlineColor
        state.label.TextStrokeTransparency = 0.5
        updateNameESPText(player, state)
    end
    -- Suppress the game's default overhead name while Echo ESP is active.
    hideDefaultNameTags(player)
    watchDefaultNameTags(player)
end
function refreshNameESP()
    if not VisualState.NameESP then
        -- Clear our ESP and restore every hidden default tag.
        for player in pairs(NameStates) do clearNameESP(player) end
        for player in pairs(NameHiddenDefaults) do restoreDefaultNameTags(player) end
        return
    end
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then updateNameESP(player) end
    end
end
local function updateNameESPDistances()
    if not VisualState.NameESP then return end
    for player, state in pairs(NameStates) do
        if player and player.Parent and state and state.gui and state.gui.Parent then
            updateNameESPText(player, state)
        end
    end
end

function clearPCLD(object)

function updatePCLD(object)

    if not VisualState.PCLD or not isPCLD(object) or not object.Parent then clearPCLD(object); return end

    local anchor = PCLDVisuals[object]

    if not anchor or not anchor.Parent then

        anchor = Instance.new("Part")

        anchor.Anchored = true

        anchor.CanCollide = false

        anchor.CanTouch = false

        anchor.CanQuery = false

        anchor.Transparency = VisualState.PCLDGlassTransparency

        anchor.Material = Enum.Material.Glass

        anchor.Color = VisualState.PCLDColor

        anchor.Size = object.Size

        anchor.CFrame = object.CFrame

        anchor.Parent = ESPFolder

        local box = Instance.new("SelectionBox")

        box.Name = "Echo_PCLD"

        box.LineThickness = 0.01

        box.Color3 = VisualState.PCLDOutlineColor

        box.SurfaceColor3 = VisualState.PCLDColor

        box.Parent = anchor

        box.Adornee = anchor

        box.Color3 = VisualState.PCLDOutlineColor

        box.SurfaceColor3 = VisualState.PCLDColor

        box.SurfaceTransparency = 1

        box.Transparency = VisualState.PCLDOutlineTransparency

        box.Visible = true

    end

end

function refreshPCLD()

    if not VisualState.PCLD then destroyMap(PCLDVisuals); return end

    for _, object in ipairs(workspace:GetDescendants()) do if isPCLD(object) then updatePCLD(object); end; end

end

function updatePCLDTracking()

    if not VisualState.PCLD then return end

    local alpha = 1 - (math.clamp(VisualState.PCLDSmoothness, 0, 1) * 0.94)

    for object, anchor in pairs(PCLDVisuals) do

        if not object.Parent or not anchor.Parent then clearPCLD(object); else

            pcall(function()

                anchor.Size = object.Size

                anchor.Color = VisualState.PCLDColor

                anchor.Transparency = VisualState.PCLDGlassTransparency

                anchor.Material = Enum.Material.Glass

                anchor.CFrame = anchor.CFrame:Lerp(object.CFrame, alpha)

            end)

        end

    end

end

function refreshSpecial(enabled, store, predicate, name, fill)

    if not enabled then destroyMap(store); return end

    local seen = {}

    for _, object in ipairs(workspace:GetDescendants()) do

        if predicate(object) then

            seen[object] = true

            local colour = name == "Echo_StickyESP" and VisualState.StickyColor or VisualState.BlobmanColor
            local outline = name == "Echo_StickyESP" and VisualState.StickyOutlineColor or VisualState.BlobmanOutlineColor
            makeHighlight(store, object, object, name, colour, 0.35, outline, 0)

        end

    end

    for object, h in pairs(store) do

        if not seen[object] or not object.Parent then pcall(function() h:Destroy() end); store[object] = nil end

    end

end

function hookESPPlayer(player)

    if player == LocalPlayer then return end

    local key = "Character_" .. player.UserId

    if ESPConnections[key] then ESPConnections[key]:Disconnect() end

    ESPConnections[key] = player.CharacterAdded:Connect(function()

        task.wait(0.35)

        clearPlayerESP(player)

        clearNameESP(player)

        updatePlayerESP(player)

        updateNameESP(player)

        -- The game recreates its default tag after our clearNameESP — hide again.
        task.delay(0.25, function()
            if VisualState.NameESP then
                hideDefaultNameTags(player)
            end
        end)

    end)

    updatePlayerESP(player)

    updateNameESP(player)

end
for _, player in ipairs(Players:GetPlayers()) do hookESPPlayer(player) end

    local key = "Character_" .. player.UserId

end)

local NameHideAccumulator = 0
ESPConnections.Render = RunService.RenderStepped:Connect(function(dt)

    updatePCLDTracking()

    if VisualState.PlayerESP then refreshPlayerESP() end

    if VisualState.NameESP then
        NameESPAccumulator = NameESPAccumulator + dt
        if NameESPAccumulator >= 0.12 then
            NameESPAccumulator = 0
            updateNameESPDistances()
        end
        -- Periodically re-hide any default tags that the game may have re-added.
        NameHideAccumulator = NameHideAccumulator + dt
        if NameHideAccumulator >= 0.25 then
            NameHideAccumulator = 0
            for player in pairs(NameStates) do
                if player and player.Parent then
                    hideDefaultNameTags(player)
                end
            end
        end
    else
        NameESPAccumulator = 0
        NameHideAccumulator = 0
    end

end)

    local Input = Options.EchoOverlayText
    if Input and Input.SetValue then
        Input:SetValue(EchoOverlayDefaultText)
    end
    EchoUpdateOverlay()
end)

task.spawn(function()
    local Stats = game:GetService("Stats")
    local StatsFPS = Stats:FindFirstChild("Workspace") and Stats.Workspace:FindFirstChild("Heartbeat")
    local StatsPing = Stats:FindFirstChild("Network") and Stats.Network:FindFirstChild("ServerStatsItem") and Stats.Network.ServerStatsItem:FindFirstChild("Data Ping")
    while EchoOverlayRunning and not Library.Unloaded do
        pcall(function()
            if StatsFPS then EchoOverlayFPS = math.round(StatsFPS:GetValue()) end
            if StatsPing then EchoOverlayPing = math.round(StatsPing:GetValue()) end
            EchoUpdateOverlay()
        end)
        task.wait(0.5)
    end
end)
Library:OnUnload(function()
    EchoOverlayRunning = false
    pcall(function()
        if EchoOverlayDragConnection then EchoOverlayDragConnection:Disconnect() EchoOverlayDragConnection = nil end
        if EchoOverlayInputConnection then EchoOverlayInputConnection:Disconnect() EchoOverlayInputConnection = nil end
        if EchoOverlayGui then EchoOverlayGui:Destroy() EchoOverlayGui = nil end
    end)
end)
