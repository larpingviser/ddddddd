local function readFile(path)
    if type(readfile) ~= "function" then return nil end
    local ok, value = pcall(readfile, path)
    if ok and type(value) == "string" then return value end
end

local function writeFile(path, value)
    if type(writefile) ~= "function" then return false end
    return pcall(writefile, path, value)
end

local function copyText(value)
    if type(setclipboard) == "function" and pcall(setclipboard, value) then return true end
    if type(toclipboard) == "function" and pcall(toclipboard, value) then return true end
    if type(writeclipboard) == "function" and pcall(writeclipboard, value) then return true end
    return false
end

local function httpGet(url)
    if game.HttpGet then

        if ok and type(body) == "string" then return true, body end
    end
    return false, "HttpGet failed"
end

local function loadSource(source)
    if type(source) ~= "table" then return false, "No source" end
    if not source.loadCode then return false, "loadCode empty" end
    local loader = getLoader()
    if not loader then return false, "Executor does not support loadstring" end
    local ok1, fn, err = pcall(loader, source.loadCode)
    if not ok1 or type(fn) ~= "function" then
        return false, "compile error: " .. tostring(err or fn)
    end
    local ok2, runErr = pcall(fn)
    if not ok2 then return false, "runtime error: " .. tostring(runErr) end
    return true, "loaded " .. tostring(source.name)
end

local function checkSource(s)
    if not s or not s.url then return "red", "No URL" end
    local ok, body = httpGet(s.url)
    if not ok then return "red", "HTTP fail" end
    if #body < 20 then return "red", "Empty" end
    local low = body:lower()
    if low:find("<html",1,true) or low:find("<!doctype",1,true) then return "red", "HTML instead of code" end
    if #body > 25000000 then return "yellow", "Very large" end
    if not body:find("end",1,true) and not low:find("loadstring",1,true) then
        return "yellow", "Unusual structure"
    end
    return "green", ("OK • %d b"):format(#body)
end

local function saveConfig(name)
    name = tostring(name or ""):gsub("^%s+",""):gsub("%s+$","")
    if name == "" then return false, "Empty name" end
    name = name:gsub("[\\/:*?\"<>|]","_")
    local p = profileFromState(); p.updated = os.time()
    configs.profiles[name] = p
    configs.active = name
    persistConfigs()
    return true
end

local function loadConfig(name, execute)
    local p = configs.profiles[name]
    if type(p) ~= "table" then return false, "Config not found" end
    applyProfile(p)
    configs.active = name
    persistConfigs()
    if execute then
        local results = {}
        for _, id in ipairs(p.loadOnActivate or {}) do
            for _, s in ipairs(SOURCES) do
                if s.id == id then
                    local ok, msg = loadSource(s)
                    table.insert(results, s.name .. ": " .. tostring(msg))
                end
            end
        end
        return true, results
    end
    return true, {}
end

local function deleteConfig(name)
    if name == "Default" then return false, "Default cannot be deleted" end
    if not configs.profiles[name] then return false, "Not found" end
    configs.profiles[name] = nil
    if configs.active == name then configs.active = "Default" end
    persistConfigs()
    return true
end

local function renameConfig(oldName, newName)
    if oldName == "Default" then return false, "Default cannot be renamed" end
    if not configs.profiles[oldName] then return false, "Not found" end
    newName = tostring(newName or ""):gsub("^%s+",""):gsub("%s+$","")
    if newName == "" then return false, "Empty name" end
    if configs.profiles[newName] then return false, "Name taken" end
    configs.profiles[newName] = configs.profiles[oldName]
    configs.profiles[oldName] = nil
    if configs.active == oldName then configs.active = newName end
    persistConfigs()
    return true
end

local function duplicateConfig(name, newName)
    if not configs.profiles[name] then return false, "Not found" end
    if configs.profiles[newName] then return false, "Name taken" end
    local ok, clone = pcall(function()
        return HttpService:JSONDecode(HttpService:JSONEncode(configs.profiles[name]))
    end)
    if not ok then return false, "Clone fail" end
    clone.updated = os.time()
    configs.profiles[newName] = clone
    persistConfigs()
    return true
end

local function exportConfig(name)
    local p = configs.profiles[name]
    if not p then return false, "Not found" end
    local payload = { format="KazbekConfig", version=VERSION, name=name, data=p }
    local text = HttpService:JSONEncode(payload)
    if copyText(text) then return true, text end
    return false, text
end

local function importConfig(text, forcedName)
    if type(text) ~= "string" or text == "" then return false, "Empty" end
    local ok, payload = pcall(function() return HttpService:JSONDecode(text) end)
    if not ok or type(payload) ~= "table" then return false, "Bad JSON" end
    local p = payload.data or payload
    if type(p) ~= "table" then return false, "Bad data" end
    local name = forcedName or payload.name or ("Imported_"..os.time())
    name = tostring(name):gsub("[\\/:*?\"<>|]","_")
    if name == "" then name = "Imported_"..os.time() end
    p.loadOnActivate = p.loadOnActivate or {}
    configs.profiles[name] = p
    configs.active = name
    persistConfigs()
    return true, name
end

local BULK_COUNTER = 0

local function parseBulkLine(line)
    line = tostring(line or ""):gsub("^%s+",""):gsub("%s+$","")
    if line == "" then return nil end
    if line:match("^https?://") then
        return {
            name = "Bulk_"..BULK_COUNTER.."_"..line:sub(-20),
            url = line,
            loadCode = ('loadstring(game:HttpGet("%s"))()'):format(line),
        }
    end
    if line:find("loadstring",1,true) then
        return { name="Bulk_"..BULK_COUNTER.."_inline", url="(inline)", loadCode=line }
    end
    return {
        name = "Bulk_"..BULK_COUNTER,
        url = line,
        loadCode = ('loadstring(game:HttpGet("%s"))()'):format(line),
    }
end

local function bulkParse(text)
    local added, skipped = 0, 0
    for line in tostring(text or ""):gmatch("[^\r\n]+") do
        BULK_COUNTER += 1
        local e = parseBulkLine(line)
        if e then
            e.id = ("bulk_%04d"):format(BULK_COUNTER)
            e.loader = e.loadCode
            e.status = "unknown"
            e.lastCheck = 0
            e.isBulk = true
            table.insert(SOURCES, e)
            added += 1
        else
            skipped += 1
        end
    end
    return added, skipped
end

local function bulkLoadAll()
    local okc, failc = 0, 0
    for _, s in ipairs(SOURCES) do
        if s.isBulk then
            local ok = loadSource(s)
            if ok then okc += 1 else failc += 1 end
            task.wait(0.1)
        end
    end
    return okc, failc
end

local function bulkAddAllToProfile()
    local prof = configs.profiles[configs.active or "Default"]
    if not prof then return false, "No profile" end
    prof.loadOnActivate = prof.loadOnActivate or {}
    local added = 0
    for _, s in ipairs(SOURCES) do
        if s.isBulk then
            local exists = false
            for _, id in ipairs(prof.loadOnActivate) do
                if id == s.id then exists = true break end
            end
            if not exists then table.insert(prof.loadOnActivate, s.id); added += 1 end
        end
    end
    persistConfigs()
    return true, added
end

local function bulkClear()
    local keep = {}
    for _, s in ipairs(SOURCES) do if not s.isBulk then table.insert(keep, s) end end
    SOURCES = keep
    BULK_COUNTER = 0
end

local FTAP_AntiLag = false

local function FTAP_StartAntiLag()
    task.spawn(function()
        while FTAP_AntiLag do
            for _, v in ipairs(workspace:GetDescendants()) do
                if not FTAP_AntiLag then break end
                if v:IsA("Beam") then
                    local a0 = v.Attachment0
                    if a0 and a0.Parent then
                        local model = a0.Parent:FindFirstAncestorOfClass("Model")
                        if model and model:FindFirstChildOfClass("Humanoid") then
                            continue
                        end
                    end
                    pcall(function() v:Destroy() end)
                end
            end
            for _, obj in ipairs(workspace:GetChildren()) do
                if not FTAP_AntiLag then break end
                if obj:IsA("BasePart") and not obj.Anchored then
                    local model = obj:FindFirstAncestorOfClass("Model")
                    if not model or not model:FindFirstChildOfClass("Humanoid") then
                        pcall(function() obj:Destroy() end)
                    end
                end
            end
            task.wait(2)
        end
    end)
end

local function FTAP_StartAntiGrab()
    task.spawn(function()
        local RS = game:GetService("ReplicatedStorage")
        local Struggle = RS:FindFirstChild("CharacterEvents") and RS.CharacterEvents:FindFirstChild("Struggle")
        while FTAP_AntiGrab do
            local char = LocalPlayer.Character
            if char then
                local head = char:FindFirstChild("Head")
                local hum = char:FindFirstChildOfClass("Humanoid")
                local hrp = char:FindFirstChild("HumanoidRootPart")
                if head and head:FindFirstChild("PartOwner") and hum and hrp then
                    pcall(function()
                        if Struggle then Struggle:FireServer(LocalPlayer) end
                    end)
                    hrp.AssemblyLinearVelocity = Vector3.zero
                    hrp.AssemblyAngularVelocity = Vector3.zero
                end
            end
            task.wait(0.1)
        end
    end)
end

local function FTAP_StartAntiKick()

local activeButtons = {}

local function bindDrag(btn, id)
    local dragging, startPos, startPosBtn, moved
    btn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            moved = false
            startPos = input.Position
            startPosBtn = btn.Position
        end
    end)
    btn.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
            if not moved then
                local action = state.buttons[id] and state.buttons[id].action
                if action and ActionRegistry[action] then
                    if action == "FTAP_AntiLag" then ActionRegistry[action](not FTAP_AntiLag)
                    elseif action == "FTAP_AntiGrab" then ActionRegistry[action](not FTAP_AntiGrab)
                    elseif action == "FTAP_AntiKick" then ActionRegistry[action](not FTAP_AntiKick)
                    else ActionRegistry[action]() end
                end
            end
        end
    end)
    UIS.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement) then
            local delta = input.Position - startPos
            if math.abs(delta.X) + math.abs(delta.Y) > 5 then moved = true end
            btn.Position = UDim2.new(
                startPosBtn.X.Scale, startPosBtn.X.Offset + delta.X,
                startPosBtn.Y.Scale, startPosBtn.Y.Offset + delta.Y
            )
            state.buttons[id].xs = btn.Position.X.Scale
            state.buttons[id].xo = btn.Position.X.Offset
            state.buttons[id].ys = btn.Position.Y.Scale
            state.buttons[id].yo = btn.Position.Y.Offset
            persistState()
        end
    end)
end

function refreshMobileButtons()
    for _, b in pairs(activeButtons) do pcall(function() b:Destroy() end) end
    activeButtons = {}
    if not IS_TOUCH then return end
    for id, cfg in pairs(state.buttons) do
        if cfg.visible ~= false then
            local btn = Instance.new("TextButton")
            btn.Name = "Kazbek_"..id
            btn.Size = UDim2.fromOffset(54, 54)
            btn.Position = UDim2.new(cfg.xs or 0, cfg.xo or 16, cfg.ys or 0.5, cfg.yo or -28)
            btn.BackgroundColor3 = Color3.fromRGB(160, 30, 40)
            btn.Text = cfg.label or "BTN"
            btn.TextColor3 = Color3.fromRGB(255,255,255)
            btn.TextSize = 11
            btn.Font = Enum.Font.GothamBold
            btn.TextWrapped = true
            btn.ZIndex = 50
            btn.Parent = ButtonsGui
            local c = Instance.new("UICorner"); c.CornerRadius = UDim.new(0, 27); c.Parent = btn
            local s = Instance.new("UIStroke"); s.Color = Color3.fromRGB(255, 120, 130); s.Thickness = 2; s.Parent = btn
            activeButtons[id] = btn
            bindDrag(btn, id)
        end
    end
end
